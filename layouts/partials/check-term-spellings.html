{{- /* 분류어 철자 충돌 점검. Hugo는 분류어(tags, categories, subjects, series)를 소문자로 바꾸고 공백을 하이픈으로 바꾼 주소로 묶는다. 그래서 "Claude Code"와 "claude-code"는 같은 /tags/claude-code/가 되고, 합쳐진 분류어의 이름은 빌드마다 먼저 읽힌 철자를 따른다. 태그 페이지 제목과 글의 article:tag 메타가 빌드마다 바뀌는 까닭이다. 같은 주소로 묶이는 철자가 둘 이상이면 철자마다 쓴 글 하나를 보이고 빌드를 멈춘다. 고칠 때는 이미 많이 쓰인 철자로 맞춘다. 뜻이 다른 말이 우연히 같은 주소가 되면 (ReAct와 React) 한쪽 이름을 바꾼다. 주소는 Hugo처럼 소문자로 바꾼 뒤 urlize로 만든다. 홈에서 partialCached로 한 번만 돈다. */ -}} {{- $problems := slice -}} {{- range $tax, $_ := site.Taxonomies -}} {{- $s := newScratch -}} {{- range $p := site.Pages -}} {{- with index $p.Params $tax -}} {{- $vals := cond (reflect.IsSlice .) . (slice .) -}} {{- range $vals -}} {{- $term := string . -}} {{- $s.SetInMap (urlize (lower $term)) $term $p.RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- range $key, $spellings := $s.Values -}} {{- if gt (len $spellings) 1 -}} {{- $parts := slice -}} {{- range $spelling, $page := $spellings -}} {{- $parts = $parts | append (printf "%q (%s)" $spelling $page) -}} {{- end -}} {{- $problems = $problems | append (printf "%s: %s" $tax (delimit $parts ", ")) -}} {{- end -}} {{- end -}} {{- end -}} {{- with $problems -}} {{- errorf "같은 주소로 합쳐지는 분류어 철자가 %d쌍 있습니다. 많이 쓰인 철자로 맞춰 주십시오.\n %s" (len .) (delimit . "\n ") -}} {{- end -}}
