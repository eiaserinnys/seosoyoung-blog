{{- $summary := "" -}} {{- $result := "" -}} {{- $raw := .RawContent -}} {{- $sections := split $raw "## 3줄 요약" -}} {{- if gt (len $sections) 1 -}} {{- $summary = index $sections 1 -}} {{- $nextHeading := findRE `(?m)^## ` $summary 1 -}} {{- with $nextHeading }}{{ $summary = index (split $summary (index . 0)) 0 }}{{ end -}} {{- $summary = replaceRE `\{\{[<%].*?[>%]\}\}` "" $summary -}} {{- $summary = strings.TrimSpace $summary -}} {{- end -}} {{- if $summary -}} {{- $html := $summary | markdownify -}} {{- $lists := findRE `(?s)<(?:ul|ol)>.*?` $html 1 -}} {{- if gt (len $lists) 0 -}} {{- $result = index $lists 0 -}} {{- else -}} {{- $result = printf "

%s

" ($html | plainify | htmlEscape) -}} {{- end -}} {{- else if .Params.summary -}} {{- $result = .Params.summary | markdownify -}} {{- else -}} {{- $result = printf "

%s

" (.Summary | plainify | htmlEscape) -}} {{- end -}} {{- return ($result | safeHTML) -}}
