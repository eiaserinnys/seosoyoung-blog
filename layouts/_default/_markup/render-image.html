{{- /* Source: PaperMod (MIT), layouts/_default/_markup/render-image.html. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $image := false -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path }} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- if and (eq .MediaType.MainType "image") (ne .MediaType.SubType "svg+xml") -}} {{- $image = . -}} {{- end -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- else -}} {{- $host := strings.ToLower $u.Hostname -}} {{- $allowed := or (eq $host "img.seosoyoung.eiaserinnys.me") (strings.HasSuffix $host ".r2.dev") -}} {{- $extension := strings.ToLower (path.Ext $u.Path) -}} {{- if and (eq $u.Scheme "https") $allowed (in (slice ".png" ".jpg" ".jpeg" ".webp" ".gif") $extension) -}} {{- $result := try (resources.GetRemote $src) -}} {{- if $result.Err -}} {{- warnf "render-image: remote image fetch failed for %q: %s" $src $result.Err -}} {{- else if not $result.Value -}} {{- warnf "render-image: remote image fetch returned no value for %q" $src -}} {{- else -}} {{- $image = $result.Value -}} {{- end -}} {{- end -}} {{- end -}} {{- $dimensions := dict -}} {{- with $image -}} {{- $dimensions = dict "width" (printf "%d" .Width) "height" (printf "%d" .Height) -}} {{- end -}} {{- $attributes := merge $dimensions .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy") -}} {{- /**/ -}}
